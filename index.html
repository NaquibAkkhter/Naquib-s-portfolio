from manim import *
import numpy as np

class HarmonicWave(Scene):
    def construct(self):
      
        title = Text("The Harmonic Wave", font_size=42, color=WHITE, weight=BOLD)
        self.play(Write(title))
        self.wait(1)
        self.play(title.animate.to_edge(UP).scale(0.8))

        
        N = 25  # 25 dots ka set
        dots = VGroup()
        
        colors = color_gradient([RED, ORANGE, YELLOW, GREEN, BLUE, PURPLE, PINK], N)
        
        tracker = ValueTracker(0)
        
         
        for i in range(N):
            # Yahi math ka jadoo hai: speed dheere-dheere badhti hai
            freq = 1.0 + i * 0.25  
            y_pos = 3 - i * (6 / (N - 1))  # Screen par upar se neeche tak set karna
            
            dot = always_redraw(
                lambda idx=i, f=freq, y=y_pos: Dot(
                    point=np.array([
                        4.0 * np.sin(f * tracker.get_value()), # X-axis par left-right ghumna
                        y, 
                        0
                    ]),
                    color=colors[idx],
                    radius=0.12
                )
            )
            dots.add(dot)
            
       
        lines = always_redraw(
            lambda: VGroup(*[
                Line(
                    dots[i].get_center(), 
                    dots[i+1].get_center(), 
                    color=colors[i], 
                    stroke_width=4
                )
                for i in range(N - 1)
            ])
        )
        
       
        self.play(FadeIn(dots), FadeIn(lines), run_time=1.5)
        self.wait(1)
        
       
        self.play(tracker.animate.set_value(8 * PI), run_time=12, rate_func=linear)
        
        self.wait(1)
        
         
        magic_text = Text("Perfect Alignment! 🤯", font_size=36, color=YELLOW)
        magic_text.next_to(dots, LEFT, buff=1)
        self.play(Write(magic_text))
        
        self.wait(3)

from manim import *
import numpy as np

class MaurerRose(Scene):
    def construct(self):
       
        title = Text("How Math Draws Flowers", font_size=42, gradient=(BLUE, PURPLE))
        subtitle = Text("The Maurer Rose (n=6, d=71°)", font_size=28, color=WHITE)
        title_group = VGroup(title, subtitle).arrange(DOWN, buff=0.2).to_edge(UP)
        
        self.play(Write(title_group))
        self.wait(0.5)

       
        n = 6     
        d = 71   
        R = 3.2  
        
        
        def get_point(k):
            theta = k * d * DEGREES
            r = R * np.sin(n * theta)
            return np.array([r * np.cos(theta), r * np.sin(theta), 0])

         
        lines = VGroup()
        colors = color_gradient([TEAL, BLUE, PURPLE, PINK, RED, YELLOW, TEAL], 361)
        
        for k in range(360):
            p1 = get_point(k)
            p2 = get_point(k + 1)
            line = Line(p1, p2, stroke_width=1.5, color=colors[k])
            lines.add(line)

       
        self.play(Create(lines, lag_ratio=1), run_time=12, rate_func=linear)
        self.wait(0.5)

         
        rose_curve = ParametricFunction(
            lambda t: np.array([
                R * np.sin(n * t) * np.cos(t),
                R * np.sin(n * t) * np.sin(t),
                0
            ]),
            t_range=[0, 2 * PI],
            color=WHITE,
            stroke_width=3
        )
        
         
        self.play(Create(rose_curve), run_time=4, rate_func=smooth)
        
        
        self.play(
            lines.animate.set_stroke(width=3), 
            rose_curve.animate.set_stroke(width=5).set_color(YELLOW),
            run_time=1, 
            rate_func=there_and_back
        )

        
        final_text = Text("Mathematics is beautiful! 🤯", font_size=36, color=PINK)
        final_text.next_to(lines, DOWN, buff=0.3)
        self.play(Write(final_text))
        
        self.wait(3)

from manim import *
import numpy as np

class TusiCouple(Scene):
    def construct(self):
        
        title = Text("Mind-Bending Math Illusion", font_size=40, color=YELLOW)
        subtitle = Text("The Tusi Couple", font_size=32, color=WHITE)
        title_group = VGroup(title, subtitle).arrange(DOWN, buff=0.2).to_edge(UP)
        
        self.play(Write(title_group), run_time=1.5)
        self.play(FadeOut(subtitle))

        
        R = 3
        N = 16  
        
         
        outer_circle = Circle(radius=R, color=TEAL, stroke_width=4)
        self.play(Create(outer_circle), run_time=1)
        
        
        lines = VGroup()
        for i in range(N):
            angle = i * PI / N
            line = Line(
                start=np.array([-R * np.cos(angle), -R * np.sin(angle), 0]),
                end=np.array([R * np.cos(angle), R * np.sin(angle), 0]),
                stroke_opacity=0.3,
                stroke_width=2
            )
            lines.add(line)
        
        self.play(Create(lines), lag_ratio=0.1, run_time=2)
        
       
        tracker = ValueTracker(0)
        
        
        dots = VGroup()
        colors = color_gradient([RED, ORANGE, YELLOW, GREEN, BLUE, PURPLE, PINK], N)
        
        for i in range(N):
            angle = i * PI / N
          
            dot = always_redraw(
                lambda idx=i, ang=angle: Dot(
                    point=np.array([
                        R * np.cos(tracker.get_value() + ang) * np.cos(ang),
                        R * np.cos(tracker.get_value() + ang) * np.sin(ang),
                        0
                    ]),
                    color=colors[idx],
                    radius=0.12
                )
            )
            dots.add(dot)

        
        
        
        self.play(FadeIn(dots[0]))
        msg1 = Text("Just a dot moving in a straight line...", font_size=24).next_to(outer_circle, DOWN, buff=0.3)
        self.play(Write(msg1))
        self.play(tracker.animate.set_value(2 * PI), run_time=3, rate_func=linear)
        
        
        self.play(FadeIn(dots[1]))
        self.play(tracker.animate.set_value(4 * PI), run_time=3, rate_func=linear)
        self.play(FadeOut(msg1))
        
        
        msg2 = Text("But watch what happens with 16 lines!", font_size=24, color=YELLOW).next_to(outer_circle, DOWN, buff=0.3)
        self.play(Write(msg2))
        self.play(FadeIn(dots[2:]))
        
        
        self.play(tracker.animate.set_value(8 * PI), run_time=6, rate_func=linear)
        
        
        msg3 = Text("They form a perfectly rolling circle! 🤯", font_size=24, color=PINK).next_to(outer_circle, DOWN, buff=0.3)
        self.play(ReplacementTransform(msg2, msg3))
        
        inner_circle = always_redraw(
            lambda: Circle(radius=R/2, color=WHITE, stroke_width=3).move_to(
                np.array([
                    (R/2) * np.cos(tracker.get_value()),
                    -(R/2) * np.sin(tracker.get_value()),
                    0
                ])
            )
        )
        
        self.play(Create(inner_circle))
        
         
        self.play(tracker.animate.set_value(14 * PI), run_time=8, rate_func=linear)
        
        self.wait(2)

from manim import *
import numpy as np

class LaserWormhole(ThreeDScene):
    def construct(self):
         
        title = Text("The 3D Laser Wormhole", font_size=36, color="#00F2FE", weight=BOLD)
        subtitle = Text("Straight Lines Creating a 3D Curve!", font_size=22, color=WHITE)
        header = VGroup(title, subtitle).arrange(DOWN, aligned_edge=LEFT, buff=0.1).to_corner(UL)
        
        self.add_fixed_in_frame_mobjects(header)
        self.play(Write(header), run_time=1.5)

        
        self.set_camera_orientation(phi=68 * DEGREES, theta=30 * DEGREES)

        
        R = 2.4       # Rings ka radius
        H = 2.2       # Upar aur neeche ki unchai (Height)
        N = 48        # Laser lines ki sankhya
        
        twist = ValueTracker(0)  # Yeh tracker cylinder ko twist karega

        
        top_ring = Circle(radius=R, color=WHITE, stroke_width=4).shift(OUT * H)
        bottom_ring = Circle(radius=R, color=WHITE, stroke_width=4).shift(IN * H)

        
        colors = color_gradient(["#00F2FE", BLUE, PURPLE, PINK, "#FF0844", YELLOW, "#00F2FE"], N)

        
        lasers = always_redraw(
            lambda: VGroup(*[
                Line(
                    start=np.array([
                        R * np.cos(2 * PI * i / N),
                        R * np.sin(2 * PI * i / N),
                        -H
                    ]),
                    end=np.array([
                        R * np.cos(2 * PI * i / N + twist.get_value()),
                        R * np.sin(2 * PI * i / N + twist.get_value()),
                        H
                    ]),
                    color=colors[i],
                    stroke_width=3.5
                )
                for i in range(N)
            ])
        )

        
        self.play(Create(top_ring), Create(bottom_ring), run_time=1.5)
        self.play(Create(lasers), run_time=2)
        
        
        self.begin_ambient_camera_rotation(rate=0.35)
        self.wait(1)

       
        self.play(twist.animate.set_value(0.75 * PI), run_time=4, rate_func=smooth)
        self.wait(1.5)

         
        self.play(twist.animate.set_value(PI), run_time=3, rate_func=smooth)
        self.wait(1)

      
        self.play(twist.animate.set_value(-0.75 * PI), run_time=5, rate_func=smooth)
        self.wait(2)

       
        self.stop_ambient_camera_rotation()
        self.play(
            FadeOut(lasers), 
            FadeOut(top_ring), 
            FadeOut(bottom_ring), 
            FadeOut(header), 
            run_time=1.5
        )
        self.wait(0.5)

from manim import *
import numpy as np

class CyberpunkPortal(ThreeDScene):
    def construct(self):
        
        title = Text("The 3D Torus Knot", font_size=36, color=BLUE, weight=BOLD)
        subtitle = Text("A Mathematical Portal", font_size=24, color=WHITE)
        group = VGroup(title, subtitle).arrange(DOWN, buff=0.1).to_corner(UL)
        
        self.add_fixed_in_frame_mobjects(group)
        self.play(Write(group), run_time=1.5)

        
        self.set_camera_orientation(phi=75 * DEGREES, theta=30 * DEGREES)

        
        def torus_knot(t):
            p = 3  # लूप की संख्या (Hole के आर-पार)
            q = 7  # ट्यूब के आर-पार घुमाव
            r_main = 3  # पोर्टल की चौड़ाई
            r_tube = 1  # मोटाई
            
            x = (r_main + r_tube * np.cos(p * t)) * np.cos(q * t)
            y = (r_main + r_tube * np.cos(p * t)) * np.sin(q * t)
            z = r_tube * np.sin(p * t)
            return np.array([x, y, z])

       
        portal = ParametricFunction(
            torus_knot,
            t_range=[0, 2 * PI],
            stroke_width=8
        )
       
        portal.set_color_by_gradient("#00F2FE", "#4FACFE", "#FF0844", "#FFB199", "#00F2FE")

         
        self.begin_ambient_camera_rotation(rate=0.4)  
        
        
        self.play(Create(portal), run_time=8, rate_func=linear)
        
        
        for _ in range(3):
            self.play(portal.animate.set_stroke(width=16), run_time=0.5, rate_func=there_and_back)

        
        self.stop_ambient_camera_rotation()
        
       
        self.play(FadeOut(portal), FadeOut(group))
        self.wait(1)

from manim import *
import numpy as np

class StringArt(Scene):
    def construct(self):
         
        title = Text("The Hidden Geometry of Numbers", font_size=42, color=YELLOW)
        self.play(Write(title))
        self.wait(1)
        self.play(title.animate.to_edge(UP).scale(0.7))

      
        N = 150  
        R = 3    
        
       
        circle = Circle(radius=R, color=WHITE, stroke_width=2)
        self.play(Create(circle), run_time=1)
        
        multiplier = ValueTracker(0) 
        
        def get_point(i):
            angle = 2 * PI * i / N + PI/2 
            return np.array([R * np.cos(angle), R * np.sin(angle), 0])
            
        
        lines = VGroup(*[Line(ORIGIN, RIGHT, stroke_width=1.5) for _ in range(N)])
        
        colors = color_gradient([TEAL, BLUE, PURPLE, PINK, RED, YELLOW], N)
        
        
        def update_lines(mobs):
            m = multiplier.get_value()
            for i, line in enumerate(mobs):
                p1 = get_point(i)               
                p2 = get_point((i * m) % N)     
                
                 
                if np.linalg.norm(p1 - p2) < 1e-5:
                    p2 = p1 + RIGHT * 0.001
                    
                line.put_start_and_end_on(p1, p2)
                line.set_color(colors[i])
                
        lines.add_updater(update_lines)
        self.add(lines)
        
        m_text = always_redraw(
            lambda: Text(f"Multiplier: {multiplier.get_value():.2f}", font_size=28, color=WHITE).to_corner(UP)
        )
        self.add(m_text)
        
         
        self.play(multiplier.animate.set_value(2), run_time=5, rate_func=smooth)
        self.wait(1)
        
        self.play(multiplier.animate.set_value(3), run_time=4, rate_func=smooth)
        self.wait(1)
        
        self.play(multiplier.animate.set_value(6), run_time=10, rate_func=linear)
        self.wait(2)
        
         
        lines.clear_updaters()
        self.play(FadeOut(lines), FadeOut(circle), FadeOut(title), FadeOut(m_text))
        
        final_text = Text("Math is Art.", font_size=45, color=PINK)
        self.play(Write(final_text))
        self.wait(2)

from manim import *
import numpy as np

class QuantumCore(ThreeDScene):
    def construct(self):
        
        title = Text("The Quantum Core", font_size=38, color=TEAL, weight=BOLD)
        subtitle = Text("3D Spherical Spirals", font_size=24, color=WHITE)
        group = VGroup(title, subtitle).arrange(DOWN, buff=0.1).to_corner(UL)
        
        self.add_fixed_in_frame_mobjects(group)
        self.play(Write(group), run_time=1.5)

        
        self.set_camera_orientation(phi=65 * DEGREES, theta=45 * DEGREES)

        
        def create_orb(radius, turns, color_grad):
            return ParametricFunction(
                lambda t: np.array([
                    radius * np.sin(t) * np.cos(turns * t),
                    radius * np.sin(t) * np.sin(turns * t),
                    radius * np.cos(t)
                ]),
                t_range=[0, PI],
                stroke_width=5
            ).set_color_by_gradient(*color_grad)

         
        orb_outer = create_orb(3.2, 25, ["#00F2FE", "#4FACFE", "#00F2FE"]) 
        # बीच की लेयर (Red / Pink)
        orb_mid = create_orb(2.2, -20, ["#FF0844", "#FFB199", "#FF0844"])   
        
        orb_inner = create_orb(1.2, 15, ["#F9D423", "#FF4E50", "#F9D423"])  

        
        self.begin_ambient_camera_rotation(rate=0.3)
        
         
        self.play(Create(orb_outer), run_time=3, rate_func=smooth)
        self.play(Create(orb_mid), run_time=2.5, rate_func=smooth)
        self.play(Create(orb_inner), run_time=2, rate_func=smooth)

        self.wait(0.5)

         
        self.play(
            orb_outer.animate.rotate(2 * PI, axis=RIGHT), # X-axis पर घूमेगा
            orb_mid.animate.rotate(2 * PI, axis=UP),      # Y-axis पर घूमेगा
            orb_inner.animate.rotate(2 * PI, axis=OUT),   # Z-axis पर घूमेगा
            run_time=6,
            rate_func=linear
        )
        
       
        self.wait(1)
        self.stop_ambient_camera_rotation()
        self.play(FadeOut(orb_outer), FadeOut(orb_mid), FadeOut(orb_inner), FadeOut(group))
        self.wait(1)
